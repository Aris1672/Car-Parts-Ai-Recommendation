server {
	server_name api2.o2plus.ru www.api2.o2plus.ru;
	charset off;
	index index.php index.html;
	disable_symlinks if_not_owner from=$root_path;
	include /etc/nginx/vhosts-includes/*.conf;
	include /etc/nginx/vhosts-resources/api2.o2plus.ru/*.conf;
	include /etc/nginx/users-resources/www-root/*.conf;
	access_log /var/www/httpd-logs/api2.o2plus.ru.access.log;
	error_log /var/www/httpd-logs/api2.o2plus.ru.error.log notice;
	set $root_path /var/www/www-root/data/www/api2.o2plus.ru;
	root $root_path;
	gzip on;
	gzip_comp_level 5;
	gzip_disable "msie6";
	gzip_types text/plain text/css application/json application/x-javascript text/xml application/xml application/xml+rss text/javascript application/javascript image/svg+xml;
	location / {
		location ~ [^/]\.ph(p\d*|tml)$ {
			try_files /does_not_exists @fallback;
		}
		location ~* ^.+\.(jpg|jpeg|gif|png|svg|js|css|mp3|ogg|mpe?g|avi|zip|gz|bz2?|rar|swf|webp|woff|woff2)$ {
			expires 24h;
			try_files $uri $uri/ @fallback;
		}
		location / {
			try_files /does_not_exists @fallback;
		}
	}
location @fallback {
		include /etc/nginx/vhosts-resources/api2.o2plus.ru/dynamic/*.conf;
		proxy_ssl_server_name on;
		proxy_pass https://car-parts-ai-recommendation.vercel.app;
		proxy_redirect off;
		proxy_set_header Host car-parts-ai-recommendation.vercel.app;
		proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
		proxy_set_header X-Forwarded-Proto $scheme;
		proxy_set_header X-Forwarded-Port $server_port;
		access_log off;
	}
	return 301 https://$host:443$request_uri;
	listen 199.189.249.4:80;
}
